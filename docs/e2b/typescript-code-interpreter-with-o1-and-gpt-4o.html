💻 核心函数实现

TypeScript 的 async/await 模式处理异步的 LLM 调用和代码执行。

matchCodeBlocks 正则提取

function matchCodeBlocks(llmResponse: string): string { const regex = /```python\n([\s\S]*?)```/g; let matches: string[] = []; let match: RegExpExecArray | null; while ((match = regex.exec(llmResponse)) !== null) { if (match[1]) matches.push(match[1]); } return matches.length > 0 ? matches.join("\n") : ""; }
